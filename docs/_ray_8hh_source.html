241 bool contains(Point const &p,
242 Real tol = Eigen::NumTraits<Real>::dummy_precision()) const {
243 Vector const v{p - m_origin};
244 Real const v_norm{v.norm()};
245 if (v_norm <= tol) {
246 return true; // The origin belongs to the ray
247 }
248 Vector const u{v / v_norm};
249 Vector const d{m_direction.normalized()};
250 Real const along{u.dot(d)};
251 // Component of u orthogonal to d (the sine of the angle between them)
252 return along >= -tol && (u - along * d).norm() <= tol;
253 }

485 bool clip(Box const &b, Real &t_entry, Real &t_exit, Real const tol) const {
486 t_entry = 0; // Only the points with t >= 0 belong to the ray
487 t_exit = std::numeric_limits<Real>::infinity();
488 for (Integer i{0}; i < N; ++i) {
489 Real const o{m_origin[i]};
490 Real const d{m_direction[i]};
491 if (std::abs(d) <= tol) {
492 // Parallel to the slab: either always inside or never
493 if (o < b.min()[i] || o > b.max()[i]) {
494 return false;
495 }
496 continue;
497 }
498 Real t_near{(b.min()[i] - o) / d};
499 Real t_far{(b.max()[i] - o) / d};
500 if (t_near > t_far) {
501 std::swap(t_near, t_far);
502 }
503 t_entry = std::max(t_entry, t_near);
504 t_exit = std::min(t_exit, t_far);
505 }
506 return t_entry <= t_exit + tol;
507 }

535 Real minimize_over_ray(Box const &b, DistanceKind const kind) const {
536 Point const &b_min{b.min()};
537 Point const &b_max{b.max()};
538 Real const infinity{std::numeric_limits<Real>::infinity()};
539
540 // Collect the breakpoints (where the ray crosses a face or a center plane)
541 std::array<Real, 2 * N + 2> breaks;
542 std::size_t n_breaks{0};
543 breaks[n_breaks++] = 0;
544 for (Integer i{0}; i < N; ++i) {
545 Real const d{m_direction[i]};
546 if (d == 0) {
547 continue;
548 }
549 if (kind == DistanceKind::INTERIOR) {
550 Real const t_1{(b_min[i] - m_origin[i]) / d};
551 Real const t_2{(b_max[i] - m_origin[i]) / d};
552 if (t_1 > 0) breaks[n_breaks++] = t_1;
553 if (t_2 > 0) breaks[n_breaks++] = t_2;
554 } else {
555 Real const t_c{(Real(0.5) * (b_min[i] + b_max[i]) - m_origin[i]) / d};
556 if (t_c > 0) breaks[n_breaks++] = t_c;
557 }
558 }
559 // Insertion sort: there are at most 2N + 1 values (std::sort on such small
560 // arrays triggers false -Warray-bounds warnings with GCC)
561 for (std::size_t k{1}; k < n_breaks; ++k) {
562 for (std::size_t h{k}; h > 0 && breaks[h] < breaks[h - 1]; --h) {
563 std::swap(breaks[h], breaks[h - 1]);
564 }
565 }
566 breaks[n_breaks++] = infinity;
567
568 // Minimize the quadratic on each interval [breaks[k], breaks[k+1]]
569 Real best_t{0};
570 Real best_value{infinity};
571 for (std::size_t k{0}; k + 1 < n_breaks; ++k) {
572 Real const t_lo{breaks[k]};
573 Real const t_hi{breaks[k + 1]};
574 if (t_hi <= t_lo) {
575 continue; // Repeated breakpoint
576 }
577 // A point strictly inside the interval selects the quadratic piece
578 Real const t_mid{t_hi < infinity ? Real(0.5) * (t_lo + t_hi) : t_lo + 1};
579
580 // Accumulate A t^2 + B t + C, each axis contributes (d t + e)^2
581 Real A{0}, B{0}, C{0};
582 for (Integer i{0}; i < N; ++i) {
583 Real const o{m_origin[i]};
584 Real const d{m_direction[i]};
585 Real const x{o + t_mid * d};
586 Real slope{0}, offset{0};
587 if (kind == DistanceKind::INTERIOR) {
588 if (x < b_min[i]) { // (m - o - d t)^2
589 slope = -d;
590 offset = b_min[i] - o;
591 } else if (x > b_max[i]) { // (o + d t - M)^2
592 slope = d;
593 offset = o - b_max[i];
594 }
595 } else {
596 Real const c{Real(0.5) * (b_min[i] + b_max[i])};
597 Real const h{Real(0.5) * (b_max[i] - b_min[i])};
598 Real const s{x >= c ? Real(1) : Real(-1)}; // (s (o + d t - c) + h)^2
599 slope = s * d;
600 offset = s * (o - c) + h;
601 }
602 A += slope * slope;
603 B += 2 * slope * offset;
604 C += offset * offset;
605 }
606
607 // Minimum of the quadratic on the interval
608 Real t{t_lo};
609 if (A > 0) {
610 t = std::clamp(-B / (2 * A), t_lo, t_hi);
611 }
612 Real const value{(A * t + B) * t + C};
613 if (value < best_value) {
614 best_value = value;
615 best_t = t;
616 }
617 }
618 return best_t;
619 }
