404 template <typename Transform> Box transformed(Transform const &t) const {
405 // Transform the center, and bound the transformed half sizes with the
406 // absolute value of the linear part (exact for the box of a rotated box)
407 Point const center{t * this->baricenter()};
408 Vector const half{t.linear().cwiseAbs() * (Real(0.5) * (m_max - m_min))};
409 return Box(Point(center - half), Point(center + half));
410 }

438 Side which_side(Real const x, Real const tol, Integer const dim) const {
439 Real const x_min{m_min(dim)};
440 Real const x_max{m_max(dim)};
441 bool const on_left{x_max < x + tol};
442 bool const on_right{x - tol < x_min};
443 if (on_left && on_right) {
444 return 0.5 * (x_min + x_max) < x ? Side::LEFT : Side::RIGHT;
445 } else if (on_left) {
446 return Side::LEFT;
447 } else if (on_right) {
448 return Side::RIGHT;
449 } else {
450 return Side::INSIDE;

528 Real squared_interior_distance(Point const &p) const {
529 Real dist2{0.0};
530 for (Integer i{0}; i < N; ++i) {
531 if (m_min[i] > p[i]) {
532 Real aux{m_min[i] - p[i]};
533 dist2 += aux * aux;
534 } else if (p[i] > m_max[i]) {
535 Real aux{p[i] - m_max[i]};
536 dist2 += aux * aux;
537 }
538 }
539 return dist2;

656 Real squared_interior_distance(Box const &b) const {
657 if (this->intersects(b))
658 return 0;
659 Real dist2{0.0};
660 for (Integer i{0}; i < N; ++i) {
661 Real aux{m_min[i] - b.m_max[i]};
662 if (aux < 0.0) {
663 aux = b.m_min[i] - m_max[i];
664 if (aux < 0.0) {
665 continue;
666 }
667 }
668 dist2 += aux * aux;
669 }
670 return dist2;

683 Real squared_interior_distance(Box const &b, Point &p1, Point &p2) const {
684 // On the axes where the boxes overlap both points are the center of the
685 // overlap (if the boxes intersect, p1 = p2 is a point of the intersection)
686 for (Integer i{0}; i < N; ++i) {
687 if (m_min[i] > b.m_max[i]) {
688 p1[i] = b.m_max[i];
689 p2[i] = m_min[i];
690 } else if (b.m_min[i] > m_max[i]) {
691 p1[i] = m_max[i];
692 p2[i] = b.m_min[i];
693 } else {
694 p1[i] = p2[i] = 0.5 * (std::min(m_max[i], b.m_max[i]) +
695 std::max(m_min[i], b.m_min[i]));
696 }
697 }
698 return (p2 - p1).squaredNorm();
699 }
