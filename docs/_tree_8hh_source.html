312 template <typename Dist>
313 void push_children_nearest_first(std::vector<DistanceIndex> &stack,
314 Node const &node, Real const bound,
315 QueryCounters &c, Dist &&dist) const {
316 c.node_checks += 2;
317 Real const d_l{dist(m_tree_structure[node.child_l].box)};
318 Real const d_r{dist(m_tree_structure[node.child_r].box)};
319 // The last pushed child is popped (visited) first
320 if (d_l <= d_r) {
321 if (d_r <= bound) stack.emplace_back(d_r, node.child_r);
322 if (d_l <= bound) stack.emplace_back(d_l, node.child_l);
323 } else {
324 if (d_l <= bound) stack.emplace_back(d_l, node.child_l);
325 if (d_r <= bound) stack.emplace_back(d_r, node.child_r);
326 }
327 }

483 void build() {
484
485 // Check if the boxes pointer is valid
486 AABBTREE_ASSERT(m_boxes_ptr != nullptr,
487 "AABBtree::Tree::build(...): The boxes pointer is "
488 "not valid. Please provide a valid pointer to the "
489 "bounding boxes before building the tree.");
490
491 BoxUniquePtrList const &boxes{*m_boxes_ptr};
492
493 // Clear tree structure
494 Integer const num_boxes{static_cast<Integer>(boxes.size())};
495 m_tree_structure.clear();
496 m_tree_boxes_map.clear();
497 m_dump_counter = 0;
498
499 // An empty set of boxes produces an empty tree
500 if (num_boxes == 0) {
501 return;
502 }
503
504 // Setup the root node
505 Node root{.box = Box{},
506 .box_long = Box{},
507 .box_ptr = 0,
508 .box_num = num_boxes,
509 .box_tot_num = num_boxes,
510 .parent = -1,
511 .child_l = -1,
512 .child_r = -1};
513
514 // Setup the boxes map and compute the root box
515 root.box.set_empty();
516 m_tree_boxes_map.resize(num_boxes);
517 for (Integer i{0}; i < num_boxes; ++i) {
518 root.box.extend(*boxes[i]);
519 m_tree_boxes_map[i] = i;
520 }
521 // The root is a leaf until it is split: all its boxes are "long" boxes
522 root.box_long = root.box;
523
524 // Build the subtrees. Every node only reorders its own range of the boxes
525 // map, so different subtrees can be built concurrently. Large subtrees
526 // are split into OpenMP tasks, small ones are built serially.
527 IndexList best_map(num_boxes);
528 BuildPart top;
529#ifdef _OPENMP
530 int const num_threads{this->openmp_threads()};
531 bool const parallel{num_threads > 1 && !omp_in_parallel() &&
532 num_boxes > AABBTREE_PARALLEL_BUILD_GRAIN};
533#pragma omp parallel num_threads(num_threads) if (parallel)
534#pragma omp single
535#endif
536 this->build_part(&top, root, &best_map, 0);
537
538 // Assemble the parts in a fixed order (the result does not depend on the
539 // number of threads)
540 m_tree_structure.reserve(2 * num_boxes + 1);
541 this->assemble_part(top, -1);
542 }

835 void depth(Integer const i, Integer &d) const {
836 d = 0;
837 if (i < 0 || i >= this->size()) {
838 return;
839 }
840 std::vector<std::pair<Integer, Integer>> stack; // (node, depth)
841 stack.reserve(128);
842 stack.emplace_back(i, 0);
843 while (!stack.empty()) {
844 auto const [id, level] = stack.back();
845 stack.pop_back();
846 Node const &node{m_tree_structure[id]};
847 if (node.child_l < 0) {
848 d = std::max(d, level);
849 } else {
850 stack.emplace_back(node.child_l, level + 1);
851 stack.emplace_back(node.child_r, level + 1);
852 }
853 }
854 }

881 b += node.box_num;

965 bool split_node(Node &node, IndexList &best_map, Integer &dumps,
966 Node &node_l, Node &node_r) {
967 BoxUniquePtrList const &boxes{*m_boxes_ptr};
968
969 // If the node has less than one object, do not split it
970 if (node.box_num <= 1) {
971 return false;
972 }
973
974 // Compute the separation line and tolerance
975 Vector const sizes{node.box.max() - node.box.min()};
976 Integer sorting[N];
977 std::iota(sorting, sorting + N, 0);
978 std::ranges::sort(sorting, std::ranges::greater{},
979 [&sizes](Integer i) { return sizes[i]; });
980
981 Integer axis, n_long, n_left, n_right, id_ini, id_end;
982 Real separation_line, separation_tolerance;
983
984 Integer n_long_saved{node.box_num + 1};
985 Integer n_diff_saved{node.box_num + 1};
986 Integer axis_saved{sorting[0]};
987 bool valid_axis_found{false};
988 for (Integer dump{0}; dump < m_max_dumpings; ++dump) {
989 axis = sorting[dump];
990 separation_line = node.box.baricenter(axis);
991 separation_tolerance = sizes[axis] * m_separation_ratio_tolerance;
992
993 // Separate short and long boxes
994 n_long = n_left = n_right = 0;
995 id_ini = node.box_ptr;
996 id_end = node.box_ptr + node.box_num;
997 while (id_ini < id_end) {
998 Box const &box_id{*boxes[m_tree_boxes_map[id_ini]]};
999 typename Box::Side const side{
1000 box_id.which_side(separation_line, separation_tolerance, axis)};
1001 switch (side) {
1002 case Box::Side::LEFT: // Left boxes are moved to the end
1003 ++n_left;
1004 --id_end;
1005 std::swap(m_tree_boxes_map[id_ini], m_tree_boxes_map[id_end]);
1006 break;
1007 case Box::Side::RIGHT: // Right boxes are moved to the end
1008 ++n_right;
1009 --id_end;
1010 std::swap(m_tree_boxes_map[id_ini], m_tree_boxes_map[id_end]);
1011 break;
1012 default:
1013 ++n_long;
1014 ++id_ini;
1015 }
1016 }
1017
1018 // Save the first and the best solution found so far
1019 Integer const n_diff{std::abs(n_left - n_right)};
1020 if ((n_long <= n_long_saved && n_long <= m_max_nodal_objects &&
1021 n_diff < n_diff_saved) ||
1022 !valid_axis_found) {
1023 n_long_saved = n_long;
1024 n_diff_saved = n_diff;
1025 axis_saved = axis;
1026 std::copy_n(m_tree_boxes_map.data() + node.box_ptr, node.box_num,
1027 best_map.data() + node.box_ptr);
1028 valid_axis_found = true;
1029 }
1030
1031 ++dumps;
1032 }
1033
1034 // Terminate splitting if the long boxes cannot be split
1035 if (n_long_saved == node.box_num) {
1036 return false;
1037 }
1038
1039 // Use the best solution found
1040 std::copy_n(best_map.data() + node.box_ptr, node.box_num,
1041 m_tree_boxes_map.data() + node.box_ptr);
1042 n_long = n_long_saved;
1043 axis = axis_saved;
1044 separation_line = node.box.baricenter(axis);
1045 separation_tolerance = sizes[axis] * m_separation_ratio_tolerance;
1046
1047 // Separate the left and right boxes
1048 n_left = n_right = 0;
1049 id_ini = node.box_ptr + n_long;
1050 id_end = node.box_ptr + node.box_num;
1051 while (id_ini < id_end) {
1052 Box const &box_id{*boxes[m_tree_boxes_map[id_ini]]};
1053 typename Box::Side const side{
1054 box_id.which_side(separation_line, separation_tolerance, axis)};
1055 switch (side) {
1056 case Box::Side::LEFT: // Left boxes stay in place
1057 ++id_ini;
1058 ++n_left;
1059 break;
1060 case Box::Side::RIGHT: // Right boxes are swapped with the last one
1061 --id_end;
1062 ++n_right;
1063 std::swap(m_tree_boxes_map[id_ini], m_tree_boxes_map[id_end]);
1064 break;
1065 default:
1066 break;
1067 }
1068 }
1069
1070 // Compute the bounding box of the long boxes, and left and right children
1071 Integer j{node.box_ptr};
1072
1073 node.box_num = n_long;
1074 node.box_long.set_empty();
1075 for (Integer i{0}; i < n_long; ++i) {
1076 node.box_long.extend(*boxes[m_tree_boxes_map[j++]]);
1077 }
1078
1079 // A new leaf owning the next `count` boxes of the map (the parent is set by
1080 // the caller)
1081 auto make_leaf = [&](Integer const count) {
1082 Integer const first{j};
1083 Box bound; // Empty box
1084 for (Integer i{0}; i < count; ++i) {
1085 bound.extend(*boxes[m_tree_boxes_map[j++]]);
1086 }
1087 return Node{.box = bound,
1088 .box_long = bound, // A leaf: all its boxes are long boxes
1089 .box_ptr = first,
1090 .box_num = count,
1091 .box_tot_num = count,
1092 .parent = -1,
1093 .child_l = -1,
1094 .child_r = -1};
1095 };
1096 node_l = make_leaf(n_left);
1097 node_r = make_leaf(n_right);
1098 return true;
1099 }

1308 template <typename Result, typename Step, typename Merge>
1309 void dual_traversal(Tree const &tree, Result &result, Workspace &ws,
1310 Step &&step, Merge &&merge) const {
1311 IndexList &stack{ws.stack};
1312 stack.clear();
1313 push_pair(stack, 0, 0);
1314#ifdef _OPENMP
1315 int const num_threads{this->openmp_threads()};
1316 bool const parallel{num_threads > 1 && !omp_in_parallel() &&
1317 this->size() + tree.size() >= AABBTREE_PARALLEL_MIN_NODES};
1318 if (parallel) {
1319 // Breadth-first expansion until there are enough pairs to balance the
1320 // load (a few rounds are enough, the number of pairs grows quickly)
1321 std::size_t const target_size{static_cast<std::size_t>(2 * 64 * num_threads)};
1322 IndexList next;
1323 for (Integer round{0}; round < 64 && !stack.empty() && stack.size() < target_size;
1324 ++round) {
1325 next.clear();
1326 for (std::size_t k{0}; k < stack.size(); k += 2) {
1327 step(stack[k], stack[k + 1], result, ws, next);
1328 }
1329 stack.swap(next);
1330 }
1331 if (stack.empty()) {
1332 return;
1333 }
1334
1335 // Depth-first traversal from each pair, in parallel
1336 IndexList const frontier{stack};
1337 stack.clear();
1338 Integer const n_pairs{static_cast<Integer>(frontier.size() / 2)};
1339 std::vector<Result> partial(n_pairs);
1340 ws.counters += this->parallel_for(n_pairs, [&](Integer i, Workspace &w) {
1341 IndexList &local_stack{w.stack};
1342 local_stack.clear();
1343 push_pair(local_stack, frontier[2 * i], frontier[2 * i + 1]);
1344 Integer ref_1, ref_2;
1345 while (!local_stack.empty()) {
1346 pop_pair(local_stack, ref_1, ref_2);
1347 step(ref_1, ref_2, partial[i], w, local_stack);
1348 }
1349 });
1350 for (Result &p : partial) {
1351 merge(result, p);
1352 }
1353 return;
1354 }
1355#else
1356 static_cast<void>(tree);
1357 static_cast<void>(merge);
1358#endif
1359 Integer ref_1, ref_2;
1360 while (!stack.empty()) {
1361 pop_pair(stack, ref_1, ref_2);
1362 step(ref_1, ref_2, result, ws, stack);
1363 }
1364 }

1371 static void push_children_pairs(IndexList &stack, Integer const ref_1,
1372 Integer const ref_2, Node const &node_1,
1373 Node const &node_2) {
1374 Integer const id_1{node_index(ref_1)};
1375 Integer const id_2{node_index(ref_2)};
1376 bool const leaf_1{is_leaf(node_1)};
1377 bool const leaf_2{is_leaf(node_2)};
1378
1379 if (leaf_1 && leaf_2) {
1380 return; // Nothing below: this branch is done
1381 }
1382 if (leaf_1) {
1383 push_pair(stack, id_1, node_2.child_l);
1384 push_pair(stack, id_1, node_2.child_r);
1385 return;
1386 }
1387 if (leaf_2) {
1388 push_pair(stack, node_1.child_l, id_2);
1389 push_pair(stack, node_1.child_r, id_2);
1390 return;
1391 }
1392
1393 // Both nodes have children: split the larger one (if it is referenced as
1394 // a whole subtree), otherwise split the other one
1395 bool const split_1{node_1.box_tot_num > node_2.box_tot_num
1396 ? !is_only_long_boxes(ref_1)
1397 : is_only_long_boxes(ref_2)};
1398 if (split_1 && !is_only_long_boxes(ref_1)) {
1399 push_pair(stack, node_1.child_l, ref_2);
1400 push_pair(stack, node_1.child_r, ref_2);
1401 if (node_1.box_num > 0 && !is_only_long_boxes(ref_2)) {
1402 push_pair(stack, only_long_boxes(id_1), id_2);
1403 }
1404 } else if (!is_only_long_boxes(ref_2)) {
1405 push_pair(stack, ref_1, node_2.child_l);
1406 push_pair(stack, ref_1, node_2.child_r);
1407 if (node_2.box_num > 0 && !is_only_long_boxes(ref_1)) {
1408 push_pair(stack, id_1, only_long_boxes(id_2));
1409 }
1410 }
1411 }

1420 static void push_self_children_pairs(IndexList &stack, Integer const ref_1,
1421 Integer const ref_2, Node const &node_1,
1422 Node const &node_2) {
1423 bool const same_node{ref_1 == ref_2 && !is_only_long_boxes(ref_1)};
1424 if (!same_node) {
1425 push_children_pairs(stack, ref_1, ref_2, node_1, node_2);
1426 return;
1427 }
1428 if (is_leaf(node_1)) {
1429 return; // A leaf with itself: nothing below
1430 }
1431 Integer const l{node_1.child_l};
1432 Integer const r{node_1.child_r};
1433 push_pair(stack, l, l);
1434 push_pair(stack, r, r);
1435 push_pair(stack, l, r);
1436 if (node_1.box_num > 0) {
1437 push_pair(stack, only_long_boxes(ref_1), l);
1438 push_pair(stack, only_long_boxes(ref_1), r);
1439 }

1541 template <typename Result>
1542 void intersect_pair_step(Tree const &tree, Integer const ref_1,
1543 Integer const ref_2, Result &result,
1544 QueryCounters &c, bool const self,
1545 IndexList &stack) const {
1546 Node const &node_1{m_tree_structure[node_index(ref_1)]};
1547 Node const &node_2{tree.m_tree_structure[node_index(ref_2)]};
1548
1549 // If the nodes do not intersect, nothing below them can
1550 ++c.node_checks;
1551 if (!node_1.box.intersects(node_2.box)) {
1552 return;
1553 }
1554
1555 // Intersect the long boxes of the two nodes
1556 if (node_1.box_num > 0 && node_2.box_num > 0) {
1557 ++c.long_box_checks;
1558 if (node_1.box_long.intersects(node_2.box_long)) {
1559 auto near = [](Box const &box, Box const &other_box_long) {
1560 return box.intersects(other_box_long);
1561 };
1562 auto visit = [&](Integer const pos_1, Box const &box_1,
1563 Integer const pos_2, Box const &box_2) {
1564 if (self && pos_1 == pos_2) {
1565 return; // A box always intersects itself
1566 }
1567 ++c.object_checks;
1568 if (box_1.intersects(box_2)) {
1569 store_pair(result, pos_1, pos_2, self);
1570 }
1571 };
1572 this->for_each_long_box_pair(tree, node_1, node_2, c, near, visit);
1573 }
1574 }
1575
1576 if (self) {
1577 push_self_children_pairs(stack, ref_1, ref_2, node_1, node_2);
1578 } else {
1579 push_children_pairs(stack, ref_1, ref_2, node_1, node_2);
1580 }
1581 }

1640 void sort_pairs_nearest_first(IndexList &stack, std::size_t const first,
1641 Tree const &tree, Real const bound) const {
1642 // At most 3 pairs are pushed per step: an insertion sort is the simplest
1643 // (std::sort on such small arrays triggers false -Warray-bounds with GCC)
1644 std::array<std::tuple<Real, Integer, Integer>, 3> pairs;
1645 std::size_t n_kept{0};
1646 for (std::size_t k{first}; k < stack.size(); k += 2) {
1647 Integer const ref_1{stack[k]};
1648 Integer const ref_2{stack[k + 1]};
1649 Real const d{this->referenced_box(ref_1).interior_distance(
1650 tree.referenced_box(ref_2))};
1651 if (d <= bound) {
1652 pairs[n_kept++] = std::make_tuple(d, ref_1, ref_2);
1653 }
1654 }
1655 for (std::size_t k{1}; k < n_kept; ++k) { // Decreasing distance
1656 for (std::size_t h{k};
1657 h > 0 && std::get<0>(pairs[h - 1]) < std::get<0>(pairs[h]); --h) {
1658 std::swap(pairs[h - 1], pairs[h]);
1659 }
1660 }
1661 stack.resize(first);
1662 for (std::size_t k{0}; k < n_kept; ++k) {
1663 push_pair(stack, std::get<1>(pairs[k]), std::get<2>(pairs[k]));
1664 }
1665 }

1672 void distance_pair_step(Tree const &tree, Integer const ref_1,
1673 Integer const ref_2, DistanceResult &result,
1674 std::atomic<Real> &global, QueryCounters &c,
1675 IndexList &stack) const {
1676 Node const &node_1{m_tree_structure[node_index(ref_1)]};
1677 Node const &node_2{tree.m_tree_structure[node_index(ref_2)]};
1678 Real bound{std::min(result.distance, global.load(std::memory_order_relaxed))};
1679
1680 // If the nodes are farther than the bound, nothing below them is closer
1681 ++c.node_checks;
1682 if (node_1.box.interior_distance(node_2.box) > bound) {
1683 return;
1684 }
1685
1686 // Compare the long boxes of the two nodes
1687 if (node_1.box_num > 0 && node_2.box_num > 0) {
1688 ++c.long_box_checks;
1689 if (node_1.box_long.interior_distance(node_2.box_long) <= bound) {
1690 auto near = [&bound](Box const &box, Box const &other_box_long) {
1691 return box.interior_distance(other_box_long) <= bound;
1692 };
1693 auto visit = [&](Integer const pos_1, Box const &box_1,
1694 Integer const pos_2, Box const &box_2) {
1695 ++c.object_checks;
1696 Real const d{box_1.interior_distance(box_2)};
1697 if (d < result.distance) { // New best pair
1698 result.candidates.clear();
1699 result.candidates[pos_1].insert(pos_2);
1700 result.distance = d;
1701 bound = std::min(bound, d);
1702 atomic_min(global, d);
1703 } else if (d == result.distance) { // Tie with the best pair
1704 result.candidates[pos_1].insert(pos_2);
1705 }
1706 };
1707 this->for_each_long_box_pair(tree, node_1, node_2, c, near, visit);
1708 }
1709 }
1710
1711 // Push the next pairs, the nearest one on top
1712 std::size_t const first{stack.size()};
1713 push_children_pairs(stack, ref_1, ref_2, node_1, node_2);
1714 this->sort_pairs_nearest_first(stack, first, tree, bound);
1715 }

1721 Real distance_tree_impl(Tree const &tree, IndexMap &candidates,
1722 Workspace &ws) const {
1723 candidates.clear();
1724 if (this->is_empty() || tree.is_empty()) {
1725 return static_cast<Real>(-1.0);

1822 template <typename Object>
1823 Real distance_impl(Object const &obj, IndexSet &candidates,
1824 Workspace &ws) const {
1825 candidates.clear();
1826 if (this->is_empty()) {
1827 return static_cast<Real>(-1.0);
1828 }
1829 BoxUniquePtrList const &boxes{*m_boxes_ptr};
1830 QueryCounters &c{ws.counters};
1831 auto distance_to = [&obj](Box const &box) {
1832 return box.interior_distance(obj);
1833 };
1834 Real distance{std::numeric_limits<Real>::infinity()}; // Best so far
1835 auto may_be_closer = [&](Box const &box) {
1836 return distance_to(box) <= distance;
1837 };
1838
1839 // Depth-first traversal visiting the nearest child first, so that the
1840 // best distance (used to prune) decreases quickly. Each node is stored
1841 // with its distance, a lower bound of the distance of its boxes.
1842 std::vector<DistanceIndex> &stack{ws.nearest_stack};
1843 stack.clear();
1844 ++c.node_checks;
1845 stack.emplace_back(distance_to(m_tree_structure[0].box), 0);
1846 while (!stack.empty()) {
1847 auto const [node_distance, id] = stack.back();
1848 stack.pop_back();
1849 if (node_distance > distance) {
1850 continue; // Nothing below this node can be closer
1851 }
1852 Node const &node{m_tree_structure[id]};
1853
1854 if (long_boxes_may_match(node, c, may_be_closer)) {
1855 Integer const end{node.box_ptr + node.box_num};
1856 for (Integer i{node.box_ptr}; i < end; ++i) {
1857 Integer const pos{m_tree_boxes_map[i]};
1858 ++c.object_checks;
1859 Real const d{distance_to(*boxes[pos])};
1860 if (d < distance) { // New closest box
1861 candidates.clear();
1862 candidates.insert(pos);
1863 distance = d;
1864 } else if (d == distance) { // Tie with the closest box
1865 candidates.insert(pos);
1866 }
1867 }
1868 }
1869
1870 if (!is_leaf(node)) {
1871 this->push_children_nearest_first(stack, node, distance, c,
1872 distance_to);
1873 }
1874 }
1875 return distance;
1876 }

1881 template <typename Object, typename Function>
1882 Real closest_impl(Object const &obj, Integer const n, IndexSet &candidates,
1883 Function &distance_function, Workspace &ws) const {
1884 candidates.clear();
1885 if (this->is_empty() || n <= 0) {
1886 return static_cast<Real>(-1.0);
1887 }
1888 BoxUniquePtrList const &boxes{*m_boxes_ptr};
1889 QueryCounters &c{ws.counters};
1890 auto distance_to = [&obj, &distance_function](Box const &box) {
1891 return distance_function(obj, box);
1892 };
1893
1894 // The best candidates found so far, in a max-heap: the farthest on top
1895 std::vector<DistanceIndex> &best{ws.heap};
1896 best.clear();
1897 auto farther = [](DistanceIndex const &a, DistanceIndex const &b) {
1898 return a.first < b.first;
1899 };
1900 std::size_t const n_wanted{static_cast<std::size_t>(n)};
1901
1902 // Pruning bound: infinite until n candidates have been found, then the
1903 // distance of the farthest candidate
1904 Real bound{std::numeric_limits<Real>::infinity()};
1905 auto may_be_closer = [&](Box const &box) { return distance_to(box) <= bound; };
1906
1907 // Depth-first traversal visiting the nearest child first
1908 std::vector<DistanceIndex> &stack{ws.nearest_stack};
1909 stack.clear();
1910 ++c.node_checks;
1911 stack.emplace_back(distance_to(m_tree_structure[0].box), 0);
1912 while (!stack.empty()) {
1913 auto const [node_distance, id] = stack.back();
1914 stack.pop_back();
1915 if (node_distance > bound) {
1916 continue; // Nothing below this node can be closer
1917 }
1918 Node const &node{m_tree_structure[id]};
1919
1920 if (long_boxes_may_match(node, c, may_be_closer)) {
1921 Integer const end{node.box_ptr + node.box_num};
1922 for (Integer i{node.box_ptr}; i < end; ++i) {
1923 Integer const pos{m_tree_boxes_map[i]};
1924 ++c.object_checks;
1925 Real const d{distance_to(*boxes[pos])};
1926 if (d >= bound) {
1927 continue;
1928 }
1929 if (best.size() == n_wanted) { // Replace the farthest candidate
1930 std::pop_heap(best.begin(), best.end(), farther);
1931 best.pop_back();
1932 }
1933 best.emplace_back(d, pos);
1934 std::push_heap(best.begin(), best.end(), farther);
1935 if (best.size() == n_wanted) {
1936 bound = best.front().first;
1937 }
1938 }
1939 }
1940
1941 if (!is_leaf(node)) {
1942 this->push_children_nearest_first(stack, node, bound, c, distance_to);
1943 }
1944 }
1945
1946 if (best.empty()) {
1947 return static_cast<Real>(-1.0);
1948 }
1949 for (DistanceIndex const &candidate : best) {
1950 candidates.insert(candidate.second);
1951 }
1952 return best.front().first; // Distance of the farthest candidate
1953 }
